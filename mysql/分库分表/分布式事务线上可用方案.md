# 方案回顾

> 架构：Laravel (PHP7) + MySQL + RabbitMQ，**本地消息表 实现最终一致性分布式事务**
>
> 业务链路：订单服务 → MQ → 库存服务 → MQ → 下游第三个服务
>
> 每个服务独立数据库：`order_db` / `stock_db` / `other_db`
>
> 核心原理：**同一个数据库事务 保存业务数据 + 写入本地消息记录**；定时任务扫描消息表投递 MQ；消费端幂等防止重复执行业务

> 环境：Laravel 5.5~7.x 适配 PHP7，使用 laravel 数据库事务、laravel 任务调度、php-amqplib 作为 rabbitmq 客户端

## 订单业务服务 app/Services/OrderService.php

```php
<?php

namespace App\Services;

use App\Models\Order;
use App\Models\LocalMsg;
use Illuminate\Support\Facades\DB;
use Ramsey\Uuid\Uuid;

/**
 * 订单服务：创建订单，写入本地消息（分布式事务：本地消息表方案）
 */
class OrderService
{
    const TOPIC_ORDER_STOCK = 'order_stock_topic'; // 发给库存服务的topic

    /**
     * 创建订单
     * @param array $params
     * [
     *   'order_no' => '',
     *   'user_id' => '',
     *   'product_id' => '',
     *   'buy_num' => ''
     * ]
     * @return Order
     * @throws \Exception
     */
    public function createOrder(array $params)
    {
        // 开启数据库本地事务：订单和消息必须同时成功或同时失败
        return DB::transaction(function () use ($params) {

            // 1.生成全局唯一消息编号
            $msgNo = Uuid::uuid4()->getHex();

            // 2.写入订单
            $order = Order::create([
                'order_no'    => $params['order_no'],
                'user_id'     => $params['user_id'],
                'product_id'  => $params['product_id'],
                'buy_num'     => $params['buy_num'],
                'status'      => 0, // 待处理
            ]);

            // 3.组装MQ消息体（订单创建事件，通知库存扣减）
            $eventData = [
                'order_no'   => $params['order_no'],
                'product_id' => $params['product_id'],
                'buy_num'    => $params['buy_num'],
                'msg_no'     => $msgNo,
            ];

            // 4.写入本地消息表，【和订单在同一个事务】，保证原子性
            LocalMsg::create([
                'msg_no'          => $msgNo,
                'topic'           => self::TOPIC_ORDER_STOCK,
                'msg_body'        => json_encode($eventData, JSON_UNESCAPED_UNICODE),
                'status'          => 0, // 0待投递
                'retry_count'     => 0,
                'next_retry_time' => date('Y-m-d H:i:s'), // 立即可以重试
            ]);

            return $order;
        });
    }
}
```

## 投递消息任务（命令行任务，laravel 定时调度）

app/Console/Commands/SendLocalMsg.php

```php
<?php

namespace App\Console\Commands;

use App\Models\LocalMsg;
use App\Services\RabbitMqService;
use Carbon\Carbon;
use Illuminate\Console\Command;

/**
 * 定时扫描本地消息表，投递MQ消息
 * 配置在Kernel.php，每3秒执行一次
 */
class SendLocalMsg extends Command
{
    protected $signature = 'msg:send';
    protected $description = '扫描本地待投递消息，发送MQ';

    protected $rabbitMqService;

    public function __construct(RabbitMqService $rabbitMqService)
    {
        parent::__construct();
        $this->rabbitMqService = $rabbitMqService;
    }

    public function handle()
    {
        // 查询：待投递消息，并且到达重试时间
        $msgList = LocalMsg::where('status', 0)
            ->where('next_retry_time', '<=', Carbon::now())
            ->get();

        foreach ($msgList as $msg) {
            try {
                // 投递消息到RabbitMQ
                $this->rabbitMqService->publish($msg->topic, $msg->msg_body);

                // 投递成功 更新状态
                $msg->status = 1;
                $msg->save();

                $this->info("消息投递成功 msgNo:{$msg->msg_no}");
            } catch (\Exception $e) {
                // 投递失败，重试次数+1，设置退避时间
                $msg->retry_count = $msg->retry_count + 1;

                // 最多重试5次，超过标记失败，人工处理
                if ($msg->retry_count >= 5) {
                    $msg->status = 2;
                }

                // 退避策略：重试次数 *10秒
                $nextTime = Carbon::now()->addSeconds($msg->retry_count * 10);
                $msg->next_retry_time = $nextTime;
                $msg->save();

                $this->error("消息投递失败 msgNo:{$msg->msg_no} err:{$e->getMessage()}");
            }
        }
    }
}
```

app/Services/RabbitMqService.php 基础 rabbitmq 封装（php7 可用）

```php
<?php

namespace App\Services;

use PhpAmqpLib\Connection\AMQPStreamConnection;
use PhpAmqpLib\Message\AMQPMessage;

/**
 * RabbitMQ基础发布服务
 */
class RabbitMqService
{
    protected $host;
    protected $port;
    protected $user;
    protected $pwd;

    public function __construct()
    {
        $this->host = env('RABBITMQ_HOST', '127.0.0.1');
        $this->port = env('RABBITMQ_PORT', 5672);
        $this->user = env('RABBITMQ_USER', 'guest');
        $this->pwd  = env('RABBITMQ_PASSWORD', 'guest');
    }

    /**
     * 发布消息
     * @param string $topic
     * @param string $body
     */
    public function publish($topic, $body)
    {
        $connection = new AMQPStreamConnection($this->host, $this->port, $this->user, $this->pwd);
        $channel = $connection->channel();
        // 声明交换机，简单topic模式
        $channel->exchange_declare($topic, 'topic', false, true, false);

        $msg = new AMQPMessage($body, [
            'delivery_mode' => AMQPMessage::DELIVERY_MODE_PERSISTENT //消息持久化
        ]);
        $channel->basic_publish($msg, $topic);

        $channel->close();
        $connection->close();
    }
}
```

## 库存消费监听脚本 （artisan 常驻消费命令）

app/Console/Commands/ConsumeOrderStock.php

```php
<?php

namespace App\Console\Commands;

use App\Models\ConsumeIdempotent;
use App\Services\StockService;
use App\Services\RabbitMqService;
use Illuminate\Console\Command;
use PhpAmqpLib\Connection\AMQPStreamConnection;
use PhpAmqpLib\Message\AMQPMessage;

/**
 * 消费订单扣库存事件
 * 命令：php artisan consume:order-stock
 */
class ConsumeOrderStock extends Command
{
    protected $signature = 'consume:order-stock';
    protected $description = '消费订单事件，执行扣库存';

    protected $stockService;

    public function __construct(StockService $stockService)
    {
        parent::__construct();
        $this->stockService = $stockService;
    }

    public function handle()
    {
        $connection = new AMQPStreamConnection(env('RABBITMQ_HOST'),5672,'guest','guest');
        $channel = $connection->channel();
        $queueName = 'order_stock_queue';
        $topic = 'order_stock_topic';

        $channel->queue_declare($queueName, false, true, false, false);
        $channel->queue_bind($queueName, $topic);

        $this->info("库存消费者启动成功，等待消息...");

        $callback = function (AMQPMessage $msg) {
            $body = $msg->getBody();
            $data = json_decode($body,true);
            $msgNo = $data['msg_no'];
            $orderNo = $data['order_no'];
            $productId = $data['product_id'];
            $buyNum = $data['buy_num'];

            try {
                // ===== 幂等判断：是否已经消费这条消息 =====
                $hasConsumed = ConsumeIdempotent::where('msg_no', $msgNo)->exists();
                if ($hasConsumed) {
                    $this->info("消息已经消费过，直接ack msgNo:{$msgNo}");
                    $msg->ack();
                    return;
                }

                // 执行业务：扣库存 + 写入本地消息，通知下游第三个服务
                $this->stockService->deductStock($productId, $buyNum, $orderNo, $msgNo);

                // 业务成功，手动ACK
                $msg->ack();
                $this->info("扣库存成功 orderNo:{$orderNo}");
            } catch (\Exception $e) {
                // 业务异常，不ack，消息重回队列，等待重试
                $this->error("消费异常:".$e->getMessage());
            }
        };

        $channel->basic_consume($queueName, '', false, false, false, false, $callback);
        while ($channel->is_consuming()) {
            $channel->wait();
        }
        $channel->close();
        $connection->close();
    }
}
```

app/Services/StockService.php

```php
<?php

namespace App\Services;

use App\Models\ConsumeIdempotent;
use App\Models\LocalMsg;
use App\Models\Stock;
use Illuminate\Support\Facades\DB;
use Ramsey\Uuid\Uuid;

/**
 * 库存服务：扣减库存，成功后发送MQ给第三个业务服务
 */
class StockService
{
    const TOPIC_STOCK_SUCCESS = 'stock_success_topic';

    /**
     * 扣库存
     * @param int $productId
     * @param int $buyNum
     * @param string $orderNo
     * @param string $msgNo MQ消息编号
     * @throws \Exception
     */
    public function deductStock($productId, $buyNum, $orderNo, $msgNo)
    {
        return DB::transaction(function () use ($productId, $buyNum, $orderNo, $msgNo) {
            // 1.写入幂等记录
            ConsumeIdempotent::create([
                'biz_no' => $orderNo,
                'msg_no' => $msgNo,
            ]);

            // 2.行锁扣库存，sql：update t_stock set stock_num=stock_num-? where product_id=? and stock_num >= ? for update
            $affected = Stock::where('product_id', $productId)
                ->where('stock_num', '>=', $buyNum)
                ->decrement('stock_num', $buyNum);

            if ($affected <= 0) {
                throw new \Exception('库存不足，扣减失败');
            }

            // 3.库存扣减成功，生成事件消息，通知第三个服务
            $newMsgNo = Uuid::uuid4()->getHex();
            $eventData = [
                'order_no'   => $orderNo,
                'product_id' => $productId,
                'buy_num'    => $buyNum,
                'msg_no'     => $newMsgNo,
            ];

            // 写入库存服务本地消息表，等待定时任务投递MQ
            LocalMsg::create([
                'msg_no'          => $newMsgNo,
                'topic'           => self::TOPIC_STOCK_SUCCESS,
                'msg_body'        => json_encode($eventData, JSON_UNESCAPED_UNICODE),
                'status'          => 0,
                'retry_count'     => 0,
                'next_retry_time' => date('Y-m-d H:i:s'),
            ]);
        });
    }
}
```

# 五、补偿逻辑（重点！当第三个服务多次失败，触发补偿）

> 场景：订单成功，库存扣成功，下游第三个服务一直消费失败，消息死信。
>
> 补偿流程：监听死信队列 → 执行库存回加接口，订单状态改为取消
>
> 库存补偿接口示例

```php
<?php
namespace App\Http\Controllers;
use App\Models\Stock;
use Illuminate\Support\Facades\DB;

class StockController
{
    /**
     * 补偿接口：回加库存
     */
    public function addStock()
    {
        $productId = request()->get('product_id');
        $num = request()->get('num');
        DB::transaction(function () use ($productId, $num) {
            Stock::where('product_id', $productId)->increment('stock_num', $num);
        });
        return ['code'=>0,'msg'=>'回加库存成功'];
    }
}
```

# 六、生产环境坑点（PHP Laravel）

1. 定时任务 `msg:send` **必须加分布式锁（redis 锁）**，多台服务器会重复投递消息
2. 消费脚本常驻进程，php 内存泄漏问题，生产配合 supervisor 托管消费进程，自动重启
3. MQ 使用**手动 ACK**，业务成功才 ack，失败不 ack；死信队列处理多次失败消息
4. 幂等表唯一索引是兜底，防止高并发重复消息
5. 数据库 for update 行锁，php7+mysql 注意事务超时时间
