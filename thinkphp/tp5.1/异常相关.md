**定义钉钉通知**

**服务类**

```php
<?php


namespace app\common\service;


use think\facade\Log;

class DingDingService
{
    public static function sendNotice($data)
    {
        $url = config('dingding.robot.dd_bot_token');
        $time = time() * 1000;//毫秒级时间戳
        $secret = config('dingding.robot.dd_bot_secret');
        if (empty($url) || empty($secret)) return true;
        $sign = hash_hmac('sha256', $time . "\n" . $secret, $secret, true);
        $sign = base64_encode($sign);
        $sign = urlencode($sign);
        $content = $data['content'] ?? "";
        $msg = [
            'msgtype' => 'text',//这是文件发送类型，可以根据需求调整
            'text' => [
                'content' => $content,
            ],
        ];
        $url = "{$url}&timestamp={$time}&sign={$sign}";
        $res = self::request_by_curl($url, $msg);
        Log::info("异常报错通知:" . $res);
    }

    public static function request_by_curl($remote_server, $post_string)
    {
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $remote_server);
        curl_setopt($ch, CURLOPT_POST, 1);
        curl_setopt($ch, CURLOPT_CONNECTTIMEOUT, 5);
        curl_setopt($ch, CURLOPT_HTTPHEADER, array('Content-Type: application/json;charset=utf-8'));
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($post_string));
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        // 线下环境不用开启curl证书验证, 未调通情况可尝试添加该代码
        curl_setopt($ch, CURLOPT_SSL_VERIFYHOST, 0);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, 0);
        $data = curl_exec($ch);
        curl_close($ch);
        return $data;
    }
}
```

**dao业务逻辑**

```php
<?php


namespace app\common\dao;


use app\common\model\SystemLogModel;

class SystemLogDao
{
    public static function create($data)
    {
        $log_level = $data['level'] ?? "info";
        $log = $data['log'] ?? "";
        $systemLog = new SystemLogModel();
        $systemLog->level = $log_level;
        if (!empty($log)) $systemLog->log = json_encode($log);
        $systemLog->ip = request()->ip();
        return $systemLog->save();
    }
    
     /**
     * 自动创建报错异常  比如中间件中的抛出异常系统日志是无法接受的所以要在中间件 异常中调用 SystemLogDao::autoCreateError($e);
     * @param $exception
     * @return bool
     */
    public static function autoCreateError($exception)
    {
        $reqUrl = request()->domain() . request()->url();
        $except_data = [
            "当前环境:" => \env('APP_ENV') == 'dev' ? "测试或开发环境" : "线上环境",
            "报错行数:" => $exception->getLine(),
            "报错代码位置:" => $exception->getFile(),
            "报错原因:" => $exception->getMessage(),
            //"报错具体原因:" => $e->getTraceAsString(),
            "报错请求地址:" => $reqUrl ?? "",
            "请求参数:" => !empty(request()->except("_user,_uuid,_org_main")) ? request()->except("_user,_uuid,_org_main") : [],
            "请求ip:" => request()->ip(),
            "请求终端:" => request()->header('user-agent'),
            "请求时间:" => date('Y-m-d H:i:s'),
        ];
        //Log::error("系统级别异常记录日志:" . json_encode($except_data));
        $sysInsertData = [
            'level' => 'error',
            'log' => $except_data
        ];
        return self::create($sysInsertData);
    }
}
```

## 优化:异常记录到数据库

### 创建数据库

**对应sql结构**

```sql
CREATE TABLE `system_except_log` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `url` text COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '请求url',
  `method` varchar(16) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '请求方式 GET,POST,PUT,PATCH,DELETE',
  `line` int NOT NULL COMMENT '报错行数',
  `params` json NOT NULL COMMENT '请求参数（已过滤敏感信息）',
  `file` text COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '异常文件完整路径',
  `message` text COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '异常简短信息',
  `ua` text COLLATE utf8mb4_unicode_ci NOT NULL COMMENT 'User-Agent 用户代理',
  `trace_as_string` text COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '异常堆栈信息',
  `ip` varchar(64) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '客户端IP，支持IPv6',
  `http_status` smallint DEFAULT NULL COMMENT 'HTTP状态码',
  `app_env` varchar(32) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '运行环境 prod/test/dev',
  `exception_type` varchar(128) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '异常类名称如PDOException',
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `system_except_log_created_at_index` (`created_at`),
  KEY `system_except_log_exception_type_index` (`exception_type`),
  KEY `system_except_log_ip_index` (`ip`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

### Service代码

>  **异常类直接调用即可**

```php
# SystemExceptLogService.php

<?php

namespace App\Services;

use App\Models\SystemExceptLog;
use App\Services\Abs\CommonService;

class SystemExceptLogService extends CommonService
{
    public function add($e)
    {
        $request=request();
        // 处理请求参数并过滤敏感信息
        $allParams = array_merge($request->query(), $request->post());
        $httpStatus = null;
        if ($e instanceof \Symfony\Component\HttpKernel\Exception\HttpExceptionInterface) {
            $httpStatus = $e->getStatusCode();
        }
        // 异常类型
        $exceptionType = get_class($e);
        // 组装原始数据源
        $data = [
            'url'             => $request->fullUrl(),
            'method'          => $request->method(),
            'line'            => $e->getLine(),
            'params'          => json_encode($allParams),
            'file'            => $e->getFile(),
            'message'         => $e->getMessage(),
            'ua'              => $request->userAgent(),
            'trace_as_string' => $e->getTraceAsString(),
            'ip'              => $request->ip(),
            'http_status'     => $httpStatus,
            'app_env'         => \env('APP_ENV'),
            'exception_type'  => $exceptionType,
        ];
        //建议优化为异步插入
        SystemExceptLog::create($data);
    }
}
```
