## 只获取字段名数组（最常用）

```php
use Illuminate\Support\Facades\Schema;

// 获取表所有字段名称，一维数组
$fields = Schema::getColumnListing('your_table_name');
```

返回示例：

```php
["id","name","phone","created_at","updated_at"]
```

> 注意：`getColumnListing` 在部分驱动下**字段是字母排序，不是建表原始顺序**。

## DB 门面写法，不引入 Schema 门面

```php
use Illuminate\Support\Facades\DB;

$fields = DB::getSchemaBuilder()->getColumnListing('your_table_name');
```

## 指定数据库连接（多库场景）

```php
// 指定连接名
$fields = DB::connection('mysql2')->getSchemaBuilder()->getColumnListing('your_table_name');
```

##  模型获取

```php
        // $model 为模型实例
        $model=new User();
        $fields = $model->getConnection()->getSchemaBuilder()->getColumnListing($model->getTable());
        dd($fields);
```

## pdo获取字段(推荐)

```php

    /**
     * 获取表名称
     * @param $tableName
     * @return array
     */
    public static function getTableText($tableName)
    {
        if (!extension_loaded('pdo')) {  //判断pdo扩展是否安装
            die('PDO extension is not loaded');
        }
        $host = config('database.connections.mysql.host');
        $dbname = config('database.connections.mysql.database');
        $username = config('database.connections.mysql.username');
        $password = config('database.connections.mysql.password');
        $port = config('database.connections.mysql.port');
        try {
            // 创建 PDO 实例
            $pdo = new \PDO("mysql:host=$host:$port;dbname=$dbname;charset=utf8", $username, $password);
            $pdo->setAttribute(\PDO::ATTR_ERRMODE, \PDO::ERRMODE_EXCEPTION);

            // 查询表的字段信息
            $query = "DESCRIBE $tableName";
            $stmt = $pdo->query($query);
            $fields = $stmt->fetchAll(\PDO::FETCH_ASSOC);

        } catch (\PDOException $e) {
            echo "Connection failed: " . $e->getMessage();
        }
        //删除时间戳字段
        if (!empty($fields) && is_array($fields)) {
            $exclude = ['id', 'created_at', 'updated_at'];
            foreach ($fields as $key => $field) {
                if (in_array($field['Field'], $exclude)) {
                    unset($fields[$key]);
                }
            }
            $fields = array_values($fields);
        }
        return $fields;
    }
```

##  获取字段有什么用实战

### 自动生成dao类

> 写代码久了就知道重复的写代码很让人烦躁 一切皆可自动化

```php
<?php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;

class AutoCreateModelDao extends Command
{
    /**
     * The name and signature of the console command.
     * 终端执行示例  php artisan create:model_dao users
     * @var string
     */
    protected $signature = 'create:model_dao {table_name}';

    /**
     * The console command description.
     *
     * @var string
     */
    protected $description = '自动生成模型对应Dao类';

    /**
     * Execute the console command.
     *
     * @return int
     */
    public function handle()
    {
        $tableName = $this->argument('table_name');
        // 转大驼峰表名，简单处理，实际建议用str_studly
        $modelName = ucfirst($tableName);
        $daoName = $modelName . 'Dao';

        $appPath = app_path();
        $dirPath = $appPath . '/Dao';
        // 目录不存在创建Dao文件夹
        if (!is_dir($dirPath)) {
            mkdir($dirPath, 0755, true);
        }

        $filePath = $dirPath . '/' . $daoName . '.php';
        if (file_exists($filePath)) {
            $this->error("Dao file already exists: {$filePath}");
            return 1;
        }

        // 获取过滤之后字段
        $fields = $this->getTableText($tableName);
        if ($fields === false) {
            $this->error('获取表字段失败！');
            return 1;
        }

        // 生成完整Dao类代码
        $daoCode = $this->generateDaoContent($daoName, $modelName, $fields);

        file_put_contents($filePath, $daoCode);
        $this->info("✅ Dao 生成成功：{$filePath}");

        return 0;
    }

    /**
     * 生成完整Dao类代码（格式化输出，PSR4 4空格缩进）
     * @param string $daoClassName
     * @param string $modelName
     * @param array $fields
     * @return string
     */
    private function generateDaoContent(string $daoClassName, string $modelName, array $fields): string
    {
        $lines = [];
        $lines[] = '<?php';
        $lines[] = '';
        $lines[] = 'namespace App\Dao;';
        $lines[] = '';
        // 引入Model
        $lines[] = "use App\\Models\\{$modelName};";
        $lines[] = '';
        $lines[] = "class {$daoClassName}";
        $lines[] = '{';

        // create静态方法
        $lines[] = '    /**';
        $lines[] = '     * 新增数据';
        $lines[] = '     * @param array $insertData';
        $lines[] = '     * @return ' . $modelName;
        $lines[] = '     */';
        $lines[] = '    public static function create(array $insertData = [])';
        $lines[] = '    {';
        $lines[] = "        \$model = new {$modelName}();";

        foreach ($fields as $field) {
            $fieldName = $field['Field'];
            $lines[] = "        \$model->{$fieldName} = \$insertData['{$fieldName}'] ?? null;";
        }

        $lines[] = '        $model->save();';
        $lines[] = '        return $model;';
        $lines[] = '    }';

        $lines[] = '';
        $lines[] = '}';

        return implode(PHP_EOL, $lines);
    }

    /**
     * 获取表字段信息，使用Laravel DB门面，不再手动PDO
     * @param string $tableName
     * @return array|false
     */
    public function getTableText(string $tableName)
    {
        try {
            $fields = DB::select("DESCRIBE `{$tableName}`");
        } catch (\Exception $e) {
            $this->error('查询表结构异常：' . $e->getMessage());
            return false;
        }

        $exclude = ['id', 'created_at', 'updated_at'];
        $result = [];
        foreach ($fields as $item) {
            $field = (array)$item;
            if (!in_array($field['Field'], $exclude)) {
                $result[] = $field;
            }
        }
        return $result;
    }
}

```
