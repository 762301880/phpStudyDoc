## 返回目录下的文件名

> 参考资料[菜鸟教程-directory](https://www.runoob.com/php/func-directory-opendir.html)

```php
/**
     * 递归扫描目录，返回所有文件名称列表
     * 自动过滤 . .. ，防御路径穿越
     * 调用示例 $files= getRecursiveFileList(__DIR__.'/../../../routes',true);
     * @param string $rootDir 根目录
     * @return array
     */
    if (!function_exists('getRecursiveFileList')) {
        function getRecursiveFileList(string $rootDir, $isFullPath = false): array
        {
            // 标准化真实路径
            $rootReal = realpath($rootDir);
            if (!$rootReal || !is_dir($rootReal)) {
                return [];
            }

            $handle = opendir($rootReal);
            if (!$handle) {
                return [];
            }

            $files = [];
            while (($entry = readdir($handle)) !== false) {
                // 跳过 . 和 ..
                if ($entry === '.' || $entry === '..') {
                    continue;
                }

                $fullPath = $rootReal . DIRECTORY_SEPARATOR . $entry;
                $realPath = realpath($fullPath);

                // 防御路径穿越：真实路径必须以根目录开头
                if ($realPath === false || strpos($realPath, $rootReal . DIRECTORY_SEPARATOR) !== 0) {
                    continue;
                }

                $isDir = is_dir($realPath);
                if (!$isDir) {
                    $files[] = $isFullPath ? $fullPath : $entry;
                } else {
                    // ✅ 关键：接收子目录递归结果，合并到数组
                    $subFiles = getRecursiveFileList($realPath,$isFullPath);
                    $files = array_merge($files, $subFiles);
                }
            }
            closedir($handle);
            return $files;
        }
    }
```
